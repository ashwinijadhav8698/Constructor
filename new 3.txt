/*
1. Question:
 Create a base class Employee with fields name and salary. Create subclasses Manager and Developer.
The manager gets a 20% bonus on salary.
The developer gets a 10% bonus.
 Write a program to calculate and display the total salary (base + bonus) for each employee.
Explanation:
 This tests how to use inheritance to share fields/methods and override logic for bonus calculations.
*/


class Employee 
{
    protected String name;
    protected double salary;

    
    public Employee(String name, double salary)
	{
        this.name = name;
        this.salary = salary;
    }

   
    public double calculateTotalSalary()
	{
        return salary; 
    }


    public void displaySalary() 
	{
        System.out.println(name + " earns: " + calculateTotalSalary());
    }
}


class Manager extends Employee 
{
    public Manager(String name, double salary) 
	{
        super(name, salary);
    }

  
    public double calculateTotalSalary() 
	{
        return salary + (0.20 * salary); 
    }
}

class Developer extends Employee 
{
    public Developer(String name, double salary)
	{
        super(name, salary);
    }

    @Override
    public double calculateTotalSalary() 
	{
        return salary + (0.10 * salary); 
    }
}

public class InheritanceDemo
 {
    public static void main(String[] args) 
	{
        Employee emp1 = new Manager("Alice", 50000);
        Employee emp2 = new Developer("Bob", 40000);

        emp1.displaySalary(); // Alice earns: 60000.0
        emp2.displaySalary(); // Bob earns: 44000.0
    }
}
